   use crate::error::Result;
   use crate::model::Model;
   use crate::tool::ToolRegistry;
   use crate::types::ModelOptions;
   use serde_json::Value;
   use std::collections::HashMap;

   /// Output of a workflow step -- always a JSON Value for uniform data flow.
   pub type StepOutput = Value;

   /// Input available to a step: outputs of all completed dependencies,
   /// keyed by dependency step name.
   pub type StepInput = HashMap<String, Value>;

   /// A single step in a workflow DAG.
   ///
   /// Steps are either LLM calls (which send a prompt to a model and optionally
   /// dispatch tool calls) or pure data transformations (which synchronously
   /// transform inputs into an output value).
   pub enum Step {
       /// An LLM call step. Builds a prompt from dependency outputs, sends it
       /// to a model, and optionally handles a single round of tool calls.
       Llm {
           /// The model to call (owned -- each step has its own model instance).
           model: Box<dyn Model>,
           /// Builds the prompt string from the outputs of this step's dependencies.
           prompt_builder: Box<dyn Fn(&StepInput) -> String + Send + Sync>,
           /// Optional tool registry for this LLM call.
           tools: Option<ToolRegistry>,
           /// Model options (temperature, max_tokens, etc.).
           options: ModelOptions,
       },
       /// A pure data transformation step. Synchronous -- no LLM or external calls.
       Transform {
           /// Transforms dependency outputs into this step's output value.
           transform: Box<dyn Fn(&StepInput) -> Result<Value> + Send + Sync>,
       },
   }
   ```

   Key design decisions to follow:
   - `Box<dyn Model>` for owned model instances (same pattern as ToolRegistry owning Box<dyn Tool>)
   - `Box<dyn Fn(...) + Send + Sync>` for closures (async-safe, trait-object compatible)
   - Transform closures are sync (not async) -- they are "pure data transformation functions"
   - `Option<ToolRegistry>` for optional tool support on LLM steps
   - Use `Result<Value>` (the framework's Result) for transform return type

use std::collections::HashMap;
use petgraph::graph::{DiGraph, NodeIndex};
use petgraph::algo::toposort;
use petgraph::Direction;

use crate::error::{Error, Result};
use super::step::Step;
```

**Workflow struct fields:**
```rust
pub struct Workflow {
    /// DAG structure. Node weights are step names (String).
    /// Edge weights are () (edges just mean "depends on").
    graph: DiGraph<String, ()>,
    /// Steps stored by name for O(1) lookup during execution.
    steps: HashMap<String, Step>,
    /// Map from step name to NodeIndex for graph operations.
    node_indices: HashMap<String, NodeIndex>,
    /// Pre-computed topological execution order (step names).
    execution_order: Vec<String>,
}
```

**`Workflow::new` constructor -- the core validation method:**

Signature: `pub fn new(steps: Vec<(String, Step)>, edges: Vec<(String, String)>) -> Result<Self>`

Edge convention: `(from, to)` means "from must complete before to" -- i.e., `from` is a dependency of `to`.

Implementation steps:
1. Check for duplicate step names. Iterate `steps`, build a `HashMap<String, Step>`. If `insert` returns `Some` (existing entry), return `Error::InvalidWorkflow(format!("duplicate step name: {}", name))`.
2. Build `DiGraph` and `node_indices` map. For each step name, `graph.add_node(name.clone())` and store the `NodeIndex` in `node_indices`.
3. Validate edges and add them to the graph. For each `(from, to)` in `edges`:
   - If `from` is not in `node_indices`, return `Error::MissingDependency { step: to.clone(), dependency: from.clone() }`.
   - If `to` is not in `node_indices`, return `Error::MissingDependency { step: from.clone(), dependency: to.clone() }`.
   - `graph.add_edge(node_indices[&from], node_indices[&to], ())`.
4. Run `toposort(&graph, None)`. On `Err(cycle)`, get the node name via `graph[cycle.node_id()]` and return `Error::InvalidWorkflow(format!("cycle detected involving step: {}", name))`.
5. Convert the `Vec<NodeIndex>` from toposort into `Vec<String>` by looking up `graph[node_idx]` for each index. Store as `execution_order`.
6. Return the validated `Workflow`.

**Accessor methods (all `&self`):**

- `pub fn execution_order(&self) -> &[String]` -- returns the pre-computed topological order.
- `pub fn step(&self, name: &str) -> Option<&Step>` -- looks up a step by name.
- `pub fn dependencies(&self, name: &str) -> Vec<String>` -- returns the names of all direct dependencies of the given step. Uses `graph.neighbors_directed(node_idx, Direction::Incoming)` to find nodes with edges pointing TO this step, then maps each to its name via `graph[dep_idx].clone()`. Returns empty Vec if step not found.
- `pub fn step_names(&self) -> Vec<&str>` -- returns all step names (useful for inspection/debugging).

**Unit tests in the same file (`#[cfg(test)] mod tests`):**

1. `test_linear_workflow_order` -- Three steps A->B->C. Assert execution_order is [A, B, C].
2. `test_diamond_workflow` -- Four steps: A->B, A->C, B->D, C->D. Assert A is first, D is last, B and C are in the middle (order between B/C is unspecified but both must come after A and before D).
3. `test_cycle_detection` -- Three steps A->B->C->A. Assert `Workflow::new` returns `Error::InvalidWorkflow` containing "cycle".
4. `test_missing_dependency` -- Edge references a step name not in the steps list. Assert `Error::MissingDependency`.
5. `test_duplicate_step_names` -- Two steps with the same name. Assert `Error::InvalidWorkflow` containing "duplicate".
6. `test_single_step_no_edges` -- One step, no edges. Assert execution_order has one entry.
7. `test_dependencies_accessor` -- Build A->B->C. Assert `dependencies("B")` returns `["A"]`, `dependencies("A")` returns `[]`, `dependencies("C")` returns `["B"]`.
